      // Prevent default only to stop any other handlers
      isNavigating = true;
      
      syncToIndex(clickedIndex);

      // Restart rotation
      clearInterval(autoRotation);
      if (isInView) {
        autoRotation = setInterval(() => {
          let nextIndex = getCurrentIndex() + 1;
          if (nextIndex >= totalTabs) nextIndex = 0;
          goToTab(nextIndex);
        }, DURATION);
      }

      setTimeout(() => {
        isNavigating = false;
      }, 100);
    });

    // Arrow navigation - capture phase to run first
    $(".testimonials_tabs-prev, .testimonials_tabs-next").on("click", function (e) {
      e.preventDefault();
      e.stopPropagation(); // Stop event from bubbling to tabs
      e.stopImmediatePropagation();

      if (isNavigating) return;

      const currentIndex = getCurrentIndex();
      const direction = $(this).hasClass("testimonials_tabs-prev") ? -1 : 1;
      let nextIndex = currentIndex + direction;

      if (nextIndex >= totalTabs) nextIndex = 0;
      if (nextIndex < 0) nextIndex = totalTabs - 1;

      goToTab(nextIndex);
    });
  };

  // Intersection Observer
  const testimonialsSection = document.querySelector(".testimonials_tabs.w-tabs");

  if (testimonialsSection) {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          isInView = entry.isIntersecting;

          if (entry.isIntersecting) {
            initializeListeners();
            startRotation();
          } else {
            stopRotation();
          }
        });
      },
      {
        threshold: 0.2,
        rootMargin: "0px"
      }
    );

    observer.observe(testimonialsSection);
  }
});

  document.addEventListener('DOMContentLoaded', () => {
    const floatingComponent = document.querySelector('.floating-video_component');
    if (!floatingComponent) return;

    const video = floatingComponent.querySelector('.floating-video_video');
    const preview = floatingComponent.querySelector('.floating-video_preview');
    const previewImg = preview?.querySelector('img');

    if (!video) return;

    // Load GIF on hover
    if (preview && previewImg?.dataset.gif) {
      preview.addEventListener('mouseenter', () => {
        previewImg.src = previewImg.dataset.gif;
      }, { once: true });
    }

    const isActive = () => floatingComponent.classList.contains('is-active');

    const openVideo = () => {
      const source = video.querySelector('source');
      if (source && source.dataset.src) {
        source.src = source.dataset.src;
        source.removeAttribute('data-src');
        video.load();
      }

      floatingComponent.classList.add('is-active');
      video.play();
    };

    const closeVideo = () => {
      floatingComponent.classList.remove('is-active', 'is-paused');
      video.pause();
      video.load();
    };

    const togglePlayPause = () => {
      if (!isActive()) return;

      if (video.paused) {
        video.play();
        floatingComponent.classList.remove('is-paused');
      } else {
        video.pause();
        floatingComponent.classList.add('is-paused');
      }
    };

    floatingComponent.addEventListener('click', (e) => {
      const target = e.target;

      if (target.closest('.floating-video_delete-button')) {
        e.stopPropagation();
        isActive() ? closeVideo() : floatingComponent.style.display = 'none';
        return;
      }

      if (target.closest('.floating-video_overlay')) {
        closeVideo();
        return;
      }

      if (target.closest('.floating-video_preview')) {
        openVideo();
        return;
      }

      if (target.closest('.floating-video_video')) {
        togglePlayPause();
      }
    });
  });
